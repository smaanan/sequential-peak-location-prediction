\section{Introduction}
%================================================

\begin{frame}{Why predict peaks?}
	\justifying
	Many environmental decisions must be taken \textbf{before} the turning point:
	\begin{itemize}
		\item Air-pollution alerts (PM\textsubscript{2.5}, NO\textsubscript{2})
		\item Heat-wave plans (temperature extremes)
		\item Peak electricity demand (cold snaps, heat waves)
		\item Flood crests (river discharge)
		\item Epidemic incidence peaks
	\end{itemize}
	\vspace{0.35cm}
	\begin{block}{Common structure}
		A noisy, unimodal time series, whose peak location $L$ and amplitude $A$
		must be inferred from observations $I(1),\ldots,I(T)$ taken
		\textbf{possibly before the peak}, $T<L$.
	\end{block}
\end{frame}


%================================================

\begin{frame}{The problem: locate the peak before you see it}
	\justifying
	\begin{equation*}
		I(t) = A\exp\!\left(-\frac{(t-L)^2}{S^2}\right) + e(t),
		\qquad t = 1,\ldots,T
	\end{equation*}
	\begin{itemize}
		\item $A$: amplitude, \quad $L$: peak location, \quad $S$: spread,
		\quad $e(t)$: additive noise.
		\item Goal: estimate $(A,L,S)$, and in particular predict $L$,
		\textbf{using only the observations up to $T$}, possibly $T < L$.
	\end{itemize}
	\vspace{0.3cm}
	\textbf{Why is this hard?}
	\begin{itemize}
		\item Noise masks the signal.
		\item ARIMA / RNN describe dependence but not the turning point.
		\item Acting too late and acting too early do not cost the same.
	\end{itemize}
\end{frame}

	\begin{center}
		\small
		Observe up to $T$; predict the peak at $L>T$ from the observed points,
		without seeing the underlying signal in the future.
	\end{center}

%================================================

\begin{frame}{Our approach: three ideas}
	\justifying
	\begin{enumerate}
		\item \textbf{Quantile generalization of least-absolute deviations.}
		Replace median regression ($\tau=\tfrac12$) by quantile regression
		($\tau\in(0,1)$) with check loss
		\[
		\rho_\tau(u)=u\bigl(\tau-\mathbf 1_{\{u<0\}}\bigr).
		\]
		\smallskip
		\textit{$\tau\to 1$: conservative peak; \ $\tau\to 0$: aggressive peak.}
		\medskip
		\item \textbf{$\tau$-aligned asymmetric Laplace noise.}
		Noise whose $\tau$-quantile is zero, so that
		\[
		-\log f_{\tau,b}(e)=\text{const}+\tfrac{1}{b}\,\rho_\tau(e).
		\]
		\smallskip
		\textit{Gives a likelihood interpretation and separates $\tau$ from SNR.}
		\medskip
		\item \textbf{Operational peak region.}
		When $\hat l(T)$ does not stabilize, build entry--exit times from the
		\emph{oscillations} of the trajectory alone — the true $L$ is never used.
	\end{enumerate}
\end{frame}

%================================================

\begin{frame}{Our contribution}
	
	We develop a quantile-based framework for early peak
	prediction, extending the weighted-median approach of
	Dermoune et al. (2026).
	
	\vspace{0.2cm}
	
	\begin{enumerate}
		
		\item \textbf{Quantile-based peak estimation}
		
		Replace the least-absolute-deviation criterion with
		a general quantile regression criterion, allowing
		different quantile levels $\tau\in(0,1)$.
		
		\vspace{0.2cm}
		
		\item \textbf{Probabilistic foundation}
		
		Introduce a $\tau$-aligned asymmetric Laplace noise
		model, linking quantile estimation to maximum
		likelihood and allowing the joint study of $\tau$
		and the signal-to-noise ratio.
		
		\vspace{0.2cm}
		
		\item \textbf{Data-driven peak-region construction}
		
		Develop a two-stage procedure based on the oscillations
		of the sequential peak-location estimates to identify
		candidate entry and exit times, without using the
		unknown true peak.
		
	\end{enumerate}
	
	\vspace{0.25cm}
	
	\begin{block}{Empirical assessment}
		Monte Carlo simulations and an application to first-wave
		COVID-19 incidence data in China, France, and Germany.
	\end{block}
	
\end{frame}

\section{Estimation method}
%================================================

\begin{frame}{Quantile estimation of the peak}
\justifying
Observations $I(t)=A\,e^{-(t-L)^2/S^2}+e(t)$, $t\le T$. Minimize over $(a,l,s)$
\[
\sum_{t=1}^{T}\rho_\tau\!\Bigl(I(t)-a\,e^{-(t-l)^2/s^2}\Bigr).
\]
\textbf{For fixed $(l,s)$:} let $w(t)=e^{-(t-l)^2/s^2}$ and $x(t)=I(t)/w(t)$. Since $w>0$,
\[
\rho_\tau\bigl(I(t)-a\,w(t)\bigr)=w(t)\,\rho_\tau\bigl(x(t)-a\bigr),
\]
so the best amplitude is the \alert{weighted $\tau$-quantile} of $x(t)$ with weights $w(t)$:
closed form, no iteration.
\vspace{0.2cm}
\begin{block}{Algorithm}
	For each integer $l\in\{1,\dots,h\}$: search $s\in[1,20]$ numerically and compute the
	profile objective; $\hat l(T)$ is its minimizer (ties: smallest $l$); then $\hat s(T)$, $\hat a(T)$.
\end{block}
\small $\tau=\tfrac12$: weighted median (Dermoune et al., 2026).
\end{frame}

%================================================

\begin{frame}{A $\tau$-aligned asymmetric Laplace noise model}
\justifying
\begin{equation*}
	f_{\tau,b}(x)=\frac{\tau(1-\tau)}{b}\,\exp\!\Bigl(-\frac{\rho_\tau(x)}{b}\Bigr),
	\qquad \mathbb P\bigl(e(t)<0\bigr)=\tau .
\end{equation*}
\begin{itemize}
	\item The $\tau$-quantile of the noise is $0$, so the quantile fit targets the signal.
	\item Negative log-likelihood $=\text{const}+\tfrac1b\sum\rho_\tau(e)$: the
	quantile estimator is the \alert{maximum likelihood estimator}.
	\item \textbf{Calibration.} $\mathrm{SNR}=A/\mathrm{sd}(e)$ is kept fixed across $\tau$ by
	\[
	b=\frac{A}{\mathrm{SNR}}\;\frac{\tau(1-\tau)}{\sqrt{\tau^2+(1-\tau)^2}} .
	\]
\end{itemize}
\vspace{0.1cm}
\small \emph{Note:} equal standard deviation does not mean equal difficulty: the information about
the $\tau$-quantile fit also depends on $\tau$.
\end{frame}

\section{Stabilization of the sequential estimate}
%================================================

\begin{frame}{Re-estimate as data arrive: the path $T\mapsto\hat l(T)$}
\justifying
\begin{columns}[T]
	\column{0.45\textwidth}
	\textbf{Stabilization time}
	\[
	T^{*}=\min\{T:\ \hat l(u)=\hat l(T)\ \ \forall u\in\{T,\dots,h\}\}
	\]
	with a plateau of at least $m=3$ horizons ($T^*=\infty$ otherwise).
	\begin{itemize}
		\item Does not use $L$.
		\item \alert{Retrospective}: defined on the whole path up to $h$, not a stopping time.
	\end{itemize}
	\small Simulation: $A=1$, $L=22$, $S=10$, $h=30$, $\tau=0.5$,
	$\mathrm{SNR}\in\{1000,100,1\}$, $M=200$.
	\column{0.53\textwidth}
	\includegraphics[width=\textwidth]{fig_tstar_examples.pdf}
\end{columns}
\end{frame}

%================================================

\begin{frame}{Stabilization: high noise changes the picture}
\centering
\begin{tabular}{ccccc}
	\toprule
	SNR & median $T^{*}$ & $\mathbb P(T^{*}<\infty)$ & $\mathbb P(T^{*}>L)$ & $\mathbb P(\hat l(T^{*})=L)$ \\
	\midrule
	1000 & 12 & 1.00 & 0.00 & 1.00 \\
	100  & 18 & 1.00 & 0.01 & 1.00 \\
	1    & 27 & 0.34 & 0.93 & 0.045 \\
	\bottomrule
\end{tabular}
\vspace{0.3cm}
\begin{itemize}
	\item \textbf{Low noise:} $\hat l(T)$ settles on $L=22$ \alert{by the time of the peak}.
	\item \textbf{High noise:} a stable plateau exists in only 34\% of series, and on $L$ in 4.5\%.
	\item The path alone does not say when its current value can be trusted.
\end{itemize}
\end{frame}

\section{Entry--exit intervals}
%================================================

\begin{frame}{When the path does not settle: look at its oscillations}
\justifying
\[
\omega_l(t,T)=\max_{t\le u,u'\le T}\bigl|\hat l(u)-\hat l(u')\bigr|
\quad\text{(non-increasing in } t\text{)}
\]
\begin{center}
	\begin{tikzpicture}[scale=0.62]
		\draw[->] (0,0) -- (11,0) node[right] {$t$};
		\draw[->] (0,0) -- (0,4.2) node[above] {$\omega_l(t,T)$};
		\draw[very thick,blue!70!black]
		(1,3.5) -- (3,3.5) -- (3,2.2) -- (5,2.2) -- (5,1) -- (8,1) -- (8,0) -- (10,0);
		\foreach \x/\l in {3/{T_{\mathrm{exit}}^{(3)}},5/{T_{\mathrm{exit}}^{(2)}},8/{T_{\mathrm{exit}}^{(1)}},10/{T}}
		\draw (\x,0.08) -- (\x,-0.08) node[below] {$\l$};
		\foreach \y/\l in {1/{\epsilon'_1},2.2/{\epsilon'_2},3.5/{\epsilon'_3}}
		\draw (0.08,\y) -- (-0.08,\y) node[left] {$\l$};
		\draw[dashed,red!70!black] (0,1.6) -- (10,1.6) node[right] {$\varepsilon$};
		\draw[dotted,red!70!black] (5.6,0) -- (5.6,1.6);
		\node[red!70!black,font=\scriptsize,anchor=south west] at (5.7,1.7)
		{$T_{\mathrm{entry}}(\varepsilon)=T_{\mathrm{exit}}^{(2)}+1$};
	\end{tikzpicture}
\end{center}
\small The strict-decrease points $T_{\mathrm{exit}}^{(1)}>T_{\mathrm{exit}}^{(2)}>\cdots$ are the
\alert{candidate exit times}; $\epsilon'_k=\omega_l(T_{\mathrm{exit}}^{(k)},T)$ is a ladder of levels.
\end{frame}

%================================================

\begin{frame}{Two-stage construction of candidate windows}
\justifying
At each horizon $T$ and for each candidate $k=0,1,\dots$:
\begin{enumerate}
	\item \textbf{Exit} $T_{\mathrm{exit}}^{(k)}(T)$ and level $\epsilon'_k$ (previous slide; $T_{\mathrm{exit}}^{(0)}=T$).
	\item \textbf{Entry:} $E_k=\{t<T_{\mathrm{exit}}^{(k)}:\ \omega_l(t,T_{\mathrm{exit}}^{(k)})\ge\epsilon'_k\}$,
	$T_k^{*}=\max E_k$, $\epsilon_k=\omega_l(T_k^{*},T_{\mathrm{exit}}^{(k)})$,
	\[
	T_{\mathrm{entry}}^{(k)}(T)=\min\{t:\ \omega_l(t,T_{\mathrm{exit}}^{(k)})=\epsilon_k\}.
	\]
\end{enumerate}
\begin{block}{Candidate peak-region windows}
	$[\,T_{\mathrm{entry}}^{(k)}(T),\,T_{\mathrm{exit}}^{(k)}(T)\,]$, built from the path only:
	\alert{the true $L$ is never used}.
\end{block}
\end{frame}

%================================================

\begin{frame}{Operational selection, and its limit}
\justifying
\begin{itemize}
	\item \textbf{Admissible set:} $\mathcal K(T)=\{k:\ T_{\mathrm{entry}}^{(k)}\le \hat l(T)\le T_{\mathrm{exit}}^{(k)}\}$.
	\item \textbf{First-admissible rule:} $\hat k(T)=\min\mathcal K(T)$; if $\mathcal K(T)=\emptyset$, wait for the next observation.
	\item $L$ is used \emph{only} to evaluate the selected interval.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Structural limit}
	$T_{\mathrm{exit}}^{(k)}\le T$, so a window can contain $L$ only once $T\ge L$:
	the intervals are \alert{not} an early warning before the peak; at high SNR the point
	estimate is already informative earlier.
\end{block}
\small No coverage guarantee: coverage is assessed by simulation.
\end{frame}

%================================================

\begin{frame}{Link with tolerance-based entry and exit times}
\justifying
For $\varepsilon\ge\varepsilon'\ge0$:
$T_{\mathrm{entry}}(\varepsilon;T)=\min\{t:\omega_l(t,T)\le\varepsilon\}$,
$T_{\mathrm{exit}}(\varepsilon';T)=\min\{t\ge T_{\mathrm{entry}}(\varepsilon;T):\omega_l(t,T)\le\varepsilon'\}$.
\begin{block}{Proposition (ladder structure)}
	\begin{enumerate}
		\item[(i)] $\epsilon'_k\le\varepsilon<\epsilon'_{k+1}\ \Rightarrow\ T_{\mathrm{entry}}(\varepsilon;T)=T_{\mathrm{exit}}^{(k+1)}(T)+1$.
		\item[(ii)] $T_{\mathrm{entry}}^{(k)}(T)=T_{\mathrm{entry}}\bigl(\epsilon_k;T_{\mathrm{exit}}^{(k)}(T)\bigr)$,
		hence $T_{\mathrm{entry}}(\epsilon_k;T)\ge T_{\mathrm{entry}}^{(k)}(T)$.
		\item[(iii)] $T^{*}=T_{\mathrm{entry}}(0;h)$ when the final plateau has $\ge m$ horizons.
	\end{enumerate}
\end{block}
\small A tolerance $\varepsilon$ selects a rung of the ladder; the two-stage construction lets the path choose.
\end{frame}

\section{Evidence}

\justifying
$M=1000$ replicates, first-admissible rule, $L=22$, $h=30$.

\section{Conclusion}
%================================================

\begin{frame}{Take-aways}
\justifying
\begin{itemize}
	\item \textbf{Quantile peak estimation:} weighted $\tau$-quantile in closed form; likelihood interpretation under $\tau$-aligned asymmetric Laplace noise.
	\item \textbf{Low noise:} the estimated peak date settles on the true date by the time of the peak.
	\item \textbf{High noise:} the path rarely settles; path-based entry--exit intervals cover the peak in 61--89\% of simulated series at $T=30$, and \alert{only after the peak}.
	\item \textbf{Ladder proposition:} ties tolerance-based stabilization rules to the two-stage construction.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Limits and next steps}
	Additive i.i.d.\ noise, single peak, no predictive benchmark yet. Next: serial dependence,
	multiplicative/count noise, forecast evaluation on environmental series
	(pollution, river discharge, electricity demand).
\end{block}
\end{frame}
